Console.WriteLine("kaç kişilik sınıf");
int ogrenci_sayisi = int.Parse(Console.ReadLine());

int[] notlar = new int[ogrenci_sayisi];
string[] ogrenci_adlari = new string[ogrenci_sayisi];


for (int i = 0; i < ogrenci_sayisi; i++)
{
    Console.WriteLine($"{i+1}. öğrencinin önce adını sonra notunu giriniz: ");
    string ad = Console.ReadLine();
    int puan = int.Parse(Console.ReadLine());
    ogrenci_adlari[i] = ad;
    notlar[i] = puan;
}

for (int i = 0; i < ogrenci_sayisi; i++)
{
    if (notlar[i] < 50)
    {
        Console.WriteLine($"{i + 1}. adı {ogrenci_adlari[i]}, notu {notlar[i]} kaldı");
        Console.WriteLine($"Geçmesi için {50 - notlar[i]} fazla alması gerekir");
    }
}
